Connected Apps
----------------

A connected app is a framework that enables an external application to integrate with Salesforce using APIs and
 standard protocols, such as SAML, OAuth, and OpenID Connect. Connected apps use these protocols to authenticate, 
authorize, and provide single sign-on (SSO) for external apps. The external apps that are integrated with Salesforce
 can run on the customer success platform, other platforms, devices, or SaaS subscriptions. For example, when you log 
in to your Salesforce mobile app and see your data from your Salesforce org, you’re using a connected app.

OAuth - User Name and password
---------------------------------

1. Authorization server is OAuth 2.0 (open Authentication)
2. Client app will send the client Id, Client Secret, UserName and Password to get request for access Token
3. Authorization server will return the Access token, this access token will work like session Id
4. Using access token we can make request to the resource in server
5. After verifying access toen server will return response

Connected with one salesforce org to another salesforce org or external system to send and get the response for the data
using rest Service

Request an Access Token
-------------------------------
To request an access token, the connected app sends the user’s username and password as
an out-of-band POST to the Salesforce token endpoint, such as the following POST.

grant_type=password&

username=testuser@salesforce.com&
password=mypassword

client_id = copy from the connected app

client_secret= copy from the connected app

username, password of the external system(like facebook or linked in or another salesforce org)

public class OAuthUserNameAndPassword {
    public string clientId {Set; get;}
    public string clientSecret {Set; get;}
    public string username {Set; get;}
    public string password {Set; get;}
    public string accessToken {Set; get;}
    public string result {Set; get;}
    public string outputResult {Set; get;}
    public string endPointUrl {Set; get;}
    public string tokenInside {Set; get;}
    public OAuthUserNameAndPassword()
    {
        //Used for authorization to get access token
        endPointUrl='https://login.salesforce.com/services/oauth2/token';
        
    }
    public void send()
    {
        Http p=new Http();
        HttpRequest req=new HttpRequest();
        req.setEndpoint(endPointUrl);
        req.setMethod('POST');
        String body='grant_type=password';
        body+='&client_id='+clientId;
        body+='&client_secret='+clientSecret;
        body+='&username='+username;
        body+='&password='+password;
        req.setBody(body);
        HttpResponse res=p.send(req);
        system.debug('res= '+res.getBody());
        result=res.getBody();
        //Get the access token from the response
        system.JSONparser jp=JSON.createParser(result);
        while(jp.nextToken()!=null)
        {
            if(jp.getText()=='access_token')
            {
                jp.nextToken();
                accessToken=jp.getText();
                //invoke();
            }
        }
   }
    public void invoke()
    {
        //Connect to external system using access token to access the rest resource
        Http hp=new Http();
        HttpRequest request=new HttpRequest();
        request.setEndPoint('https://auradomai23n-dev-ed.my.salesforce.com/services/apexrest/DisplayMessage');
        //Above end point store it in remote site settings and DisplayMessage is the rest resource in the external system
        request.setHeader('Authorization','Bearer '+accessToken);
        //We will receive the Bearer token type and dont forget to give one space after bearer  
        request.setMethod('GET');
        HttpResponse res=hp.send(request); 
        system.debug('getStatus= '+res.getStatus()); 
        system.debug('res= '+res.getBody());
        outputResult=res.getBody();
        
    }
    
    
}

@RestResource(urlMapping='/MyContact/v1/*')
global class ContactHandler {

        global String FirstName;	//asdada

        global String Phone;	//324244

    @HttpPost
    global  static String createContact(String firstName,String lastName,String Phone){
        String result;
        system.JSONGenerator js= JSON.createGenerator(false);
        js.writeStartObject();
        
        try{
            Contact con=new Contact();
            con.FirstName=firstName;
            con.LastName=lastName;
            con.Phone=Phone;
            insert con;
            
            js.writeStringField('Status','Success');
            js.writeStringField('Id',con.Id);
            //js.writeObjectField('Contact',con);
            js.writeFieldName('Contact');
            js.writeStartObject();
            js.writeStringField('Id', con.Id);
            js.writeStringField('FirstName', con.FirstName);
            js.writeStringField('LastName', con.LastName);
            js.writeStringField('Phone', con.Phone);
            js.writeEndObject();
            
            
        }
        catch(Exception e){
            js.writeStringField('Status','Failed');
            js.writeStringField('ErrorMessage',e.getMessage());
        }
        js.writeEndObject();
        result=js.getAsString();
        return result;
        
    }
    
    /*
* Example Json -
{"contactWrapper":
{"Id" :"0030o00003WFMgVAAX","FirstName" :"Park","LastName" :"Avenue", "Phone" :"8765433222"}}
*/
    @HttpPatch
    global static String updateContact(cls_Contact contactWrapper){
        String result;
        Boolean isValidContact=false;
        RestRequest req=RestContext.request;
        RestResponse res=RestContext.response;
        String contactId=contactWrapper.Id;
        system.debug('contactId='+contactId);
        system.JSONGenerator js= JSON.createGenerator(true);
        js.writeStartObject();
        
        try{
            if(contactId!=null && contactId!=''){
                
                Contact con=[SELECT Id, FirstName, LastName,Phone FROM Contact where Id=:contactId];
                if(con!=null){               
                    con.FirstName=contactWrapper.FirstName;
                    con.LastName=contactWrapper.LastName;
                    con.Phone=contactWrapper.Phone;
                    update con;
                    js.writeStringField('Status','Contact Successfully updated');
                    isValidContact=true;
                    
                }
            }
            if(!isValidContact){
                js.writeStringField('Status','Not a valid contact Id');
            }
            
        }
        catch(Exception e){
            js.writeStringField('Status','Failed');
            js.writeStringField('ErrorMessage',e.getMessage());
        }
        js.writeEndObject();
        result=js.getAsString();
        return result;
        
    }
    
    @HttpPut
    global static String upsertContact(cls_Contact contactWrapper){
        String result;
        Boolean isValidContact=false;
        system.JSONGenerator js= JSON.createGenerator(true);
        js.writeStartObject();
       
            Contact con=new Contact();
            con.Id=contactWrapper.Id;
            con.FirstName=contactWrapper.FirstName;
            con.LastName=contactWrapper.LastName;
            con.Phone=contactWrapper.Phone;
            List<Contact> conList=new List<Contact>{con};
             
            List<Database.UpsertResult> upsertResults=Database.upsert(conList, false);
            for(Database.UpsertResult upResult:upsertResults){
                if(upResult.isSuccess()){
                    isValidContact=true;
                    if(upResult.isCreated()){
                         js.writeStringField('Status','Contact inserted successfully');
                    }
                    else
                    {
                        js.writeStringField('Status','Contact Updated successfully');
                    }
                }
                else
                {
                    List<Database.Error> errors=upResult.getErrors();
                    for(Database.Error err:errors){
                        js.writeStringField('Status','Failed');
                        js.writeStringField('ErrorMessage',+err.getMessage());
                    }
                }
            }
            
            
            
        js.writeEndObject();
        result=js.getAsString();
        return result;
        
    }
    
    @HttpDelete
    global  static String deleteContact(){
        String result;
        RestRequest req=RestContext.request;
        RestResponse res=RestContext.response;
        Map<String,String> inputs=req.params;
        Id deletedContactId;
        String firstName=inputs.get('firstName');
        system.JSONGenerator js= JSON.createGenerator(true);
        js.writeStartObject();
        
        try{
            if(firstName!=null && firstName!=''){
                Contact con=[SELECT Id FROM Contact where FirstName=:firstName limit 1];
                deletedContactId=con.Id;
                delete con;
                js.writeStringField('Status','Deleted Successfully');
                js.writeStringField('Id',deletedContactId);
                js.writeObjectField('Contact',con);
            }
            
        }
        catch(Exception e){
            js.writeStringField('Status','Failed');
            js.writeStringField('ErrorMessage',e.getMessage());
        }
        js.writeEndObject();
        result=js.getAsString();
        system.debug('statusCode='+res.statusCode);
        system.debug('headers='+res.headers);
        return result;
        
    }
