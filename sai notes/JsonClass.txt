-----

serialize(objectToSerialize, suppressApexObjectNulls)
-------------------------------------------------------
Suppresses null values when serializing Apex objects into JSON content.
Signature
public static String serialize(Object objectToSerialize, Boolean suppressApexObjectNulls)

Parameters
objectToSerialize
Type: Object
The Apex object to serialize.
suppressApexObjectNulls
Type: Boolean
If true, remove null values before serializing the JSON object.

public class JsonExample1 {

    public static String createAccount(String name,String description){
        System.JSONGenerator js=JSON.createGenerator(true);
        String result;
        js.writeStartObject();
        try{
            Account acc=new Account();
            acc.Name=name;
            acc.Description=description;
            insert acc;
            js.writeIdField('Id', acc.Id);
            js.writeStringField('Status','Success');
        }
        catch(Exception e){
            js.writeStringField('Error', e.getMessage());
            js.writeStringField('Status','Failed');
        }
        js.writeEndObject();
        result=js.getAsString();
        
        return result;
        
    }
    
}

public class JsonExample2 {
    public static String searchAccount(String searchKey){
     
       /* {status:, count:, Account:[{},{},.....{}]}
            {status:, count:, Account:{}}
        */
        String searchText=searchKey+' %';
        String result;
        system.debug('searchText='+searchText);
        List<Account> accountList=[SELECT Id, Name, Type FROM Account where Name like :searchText limit 5];
        system.debug('accountList='+accountList);
        system.JSONGenerator js= JSON.createGenerator(true);
        js.writeStartObject();
        
        Integer accountSize=accountList.size();
        if(accountSize > 0){
            js.writeStringField('Status','Success');
            
            js.writeNumberField('count', accountSize);
            js.writeFieldName('Account');
            js.writeStartArray();
            for(Account acc:accountList){
                js.writeObject(acc);                
            }
            js.writeEndArray();
            
        }
        else
        {
            js.writeStringField('Status','failure');            
            js.writeNumberField('count', 0);
            
        }
        js.writeEndObject();
        result=js.getAsString();
        return result;
        
    }
}

public class JsonExample4 {
    @future(callout=true)
    public static void parseJSONResponse() {   
        Double grandTotal=0;
        String key;
        Double listPrice;
        List<lineItem> myListItem=new List<lineItem>();
        Http hp=new Http();
        HttpRequest req=new HttpRequest();
        String endpoint='https://docsample.herokuapp.com/jsonSample';
        req.setEndpoint(endpoint);
        req.setMethod('GET');
        HttpResponse res=hp.send(req);
        system.debug('getStatus='+res.getStatus());
        system.debug('getStatusCode='+res.getStatusCode());
        system.debug('getHeaderKeys='+res.getHeaderKeys());
        system.debug('getHeaderKeys='+res.getBody());
        String jsonString =res.getBody();
        if(jsonString.length() > 0){
            System.JSONParser jp=JSON.createParser(jsonString);
            
            while(jp.nextToken()!=null){
                key=jp.getText();
                if(jp.getCurrentToken()==JSONToken.FIELD_NAME && jp.getText()=='totalPrice'){
                    jp.nextToken();
                    grandTotal =grandTotal +jp.getDoubleValue();
                    
                }
                //system.debug('key');
                if(key == 'lineItems'){
                    jp.nextToken();
                    while(jp.nextToken() != JSONToken.END_ARRAY){
                        listPrice=0;
                        lineItem li=(lineItem)jp.readValueAs(lineItem.class);
                        listPrice=li.UnitPrice * li.Quantity;
                        system.debug('listPrice='+listPrice);
                        lineItem currentLineItem=new lineItem();
                        currentLineItem.UnitPrice=li.UnitPrice;
                        currentLineItem.Quantity=li.Quantity;
                        currentLineItem.ProductName=li.ProductName;
                        currentLineItem.listPrice=listPrice;
                        myListItem.add(currentLineItem);
                    }
                }
            }
             system.debug('grandTotal='+grandTotal);
            system.debug('myListItem='+myListItem);
            
        }
       
        
        
    }
    
    public class lineItem{
        Public Double UnitPrice;
        Public Double Quantity;
        public String ProductName;
        public Double listPrice;
    }
    
}
